# hyprminimal runtime: official Arch repos only (core/extra). `hyprminimal status` and install.sh check these.
# install.sh installs these plus the base desktop (see its BASE list).

# Desktop UI
hyprland                  # The compositor.
waybar                    # Status bar: workspace, clock, CPU, RAM, battery, network, volume.
rofi                      # Application launcher, search menu, and custom application menus.

# Hyprland session management
hyprlock                  # Lock screen when manually locking the session.
greetd                    # Login daemon on tty1 (bin/hyprminimal-greeter).
cage                      # Tiny kiosk compositor the login screen runs in.
json-c                    # greetd messages in greeter/greeter.c (built at install).
hyprsunset                # Night light: reduces blue light by adjusting display color temperature.

# Settings app (Python + GTK4); popups, control center, notifications and OSD are C (helpers/panels.c)
python                    # Settings app, launcher.
python-gobject            # GTK bindings for Python.
gtk4                      # Toolkit (Settings, and headers for helpers/panels.c).
libadwaita                # Settings window widgets.
gtk4-layer-shell          # Popups, notifications and OSD attached to the screen edge.
gcc                       # Builds the small C helpers (helpers/).
pkgconf                   # Finds gtk4 flags for helpers/panels.c.

# Media and system utilities
playerctl                 # Control media players: play, pause, next, previous, volume.
jq                        # Parse and manipulate JSON from scripts and command output.
brightnessctl             # Control laptop screen backlight brightness.
libnotify                 # Notification library used by applications and notify-send.
pavucontrol               # GUI audio mixer for selecting and adjusting audio devices.
wireplumber               # wpctl: volume and mute.
power-profiles-daemon     # Battery popup: power modes.
kitty                     # Terminal (launcher "rm NAME" runs in it).

# Network and Bluetooth
networkmanager            # nmcli: Wi-Fi panel and control center.
dnsmasq                   # DHCP/DNS for the Wi-Fi hotspot (NetworkManager shared mode).
iw                        # Hotspot: second (access-point) interface so Wi-Fi stays connected.
hostapd                   # Hotspot next to the Wi-Fi connection (scripts/hotspot-iface).
network-manager-applet    # System tray interface for NetworkManager and Wi-Fi connections.
bluez-utils               # bluetoothctl: Bluetooth toggle.
blueman                   # Bluetooth manager for pairing and managing Bluetooth devices.

# Fonts and icons
noto-fonts                # General-purpose fonts with broad Unicode coverage.
ttf-nerd-fonts-symbols    # Nerd Font symbols and icons for Waybar and other UI components.
papirus-icon-theme        # App icons in the launcher.
qt6ct                     # Dark theme for Qt apps (QT_QPA_PLATFORMTHEME=qt6ct).
